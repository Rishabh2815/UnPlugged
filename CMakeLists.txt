cmake_minimum_required(VERSION 3.20)
project(upi_offline_mesh LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenSSL REQUIRED)
find_package(Boost REQUIRED COMPONENTS system)
add_executable(upi-mesh src/main.cpp)
target_include_directories(upi-mesh PRIVATE ${Boost_INCLUDE_DIRS})
target_link_libraries(upi-mesh PRIVATE OpenSSL::Crypto Boost::system)
target_compile_options(upi-mesh PRIVATE -Wall -Wextra -Wpedantic)
